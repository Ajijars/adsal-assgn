
#include <iostream>
#include <cstring>
using namespace std;

char s1[100], s2[100];

int canTransform(int i, int j) {
    if (s1[i] == '\0' && s2[j] == '\0')
        return 1;

    if (s1[i] == '\0')
        return 0;

    if (s1[i] >= 'a' && s1[i] <= 'z') {
        if (s2[j] != '\0' && s1[i] - 32 == s2[j]) {
            if (canTransform(i + 1, j + 1))
                return 1;
        }

        return canTransform(i + 1, j);
    }

    if (s2[j] != '\0' && s1[i] == s2[j])
        return canTransform(i + 1, j + 1);

    return 0;
}

int main() {
    cout << "Enter first string: ";
    cin >> s1;

    cout << "Enter second string: ";
    cin >> s2;

    if (canTransform(0, 0))
        cout << "YES, transformation is possible.";
    else
        cout << "NO, transformation is not possible.";

    return 0;
}
