
#include <iostream>
using namespace std;

int coins[20], current[100], count = 0;

void findCombinations(int sum, int index, int n, int N) {
    if (sum == 0) {
        count++;
        cout << "{ ";
        for (int i = 0; i < n; i++)
            cout << current[i] << " ";
        cout << "}" << endl;
        return;
    }

    for (int i = index; i < N; i++) {
        if (coins[i] <= sum) {
            current[n] = coins[i];
            findCombinations(sum - coins[i], i, n + 1, N);
        }
    }
}

int main() {
    int N, sum;

    cout << "Enter number of coins: ";
    cin >> N;

    cout << "Enter coins: ";
    for (int i = 0; i < N; i++)
        cin >> coins[i];

    cout << "Enter target sum: ";
    cin >> sum;

    cout << "Possible combinations:" << endl;
    findCombinations(sum, 0, 0, N);

    cout << "Total number of ways = " << count << endl;

    return 0;
}
