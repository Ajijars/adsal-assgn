
#include <iostream>
using namespace std;

int arr[20], subset[20], n;

int subsetSum(int i, int sum, int k, int &len) {
    if (sum == 0) {
        len = k;
        return 1;
    }

    if (i == n || sum < 0)
        return 0;

    subset[k] = arr[i];

    if (subsetSum(i + 1, sum - arr[i], k + 1, len))
        return 1;

    return subsetSum(i + 1, sum, k, len);
}

int main() {
    int sum, len = 0;

    cout << "Enter number of elements: ";
    cin >> n;

    cout << "Enter array elements: ";
    for (int i = 0; i < n; i++)
        cin >> arr[i];

    cout << "Enter target sum: ";
    cin >> sum;

    if (subsetSum(0, sum, 0, len)) {
        cout << "Subset: ";
        for (int i = 0; i < len; i++)
            cout << subset[i] << " ";
    } else {
        cout << "No subset found";
    }

    return 0;
}
